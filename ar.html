Function R8H_NormalizeText(ByVal s As String) As String Dim t As String t = CStr(s) On Error Resume Next t = StrConv(t, vbNarrow) On Error GoTo 0 t = UCase$(t) t = Replace(t, vbCr, "") t = Replace(t, vbLf, "") t = Replace(t, "　", "") t = Replace(t, " ", "") t = Replace(t, "･", "/") t = Replace(t, "・", "/") t = Replace(t, "／", "/") R8H_NormalizeText = t End Function
